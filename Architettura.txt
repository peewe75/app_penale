function updateCurrentTime(time) { const m = Math.floor(time/60).toString().padStart(2,'0'), s = Math.floor(time%60).toString().padStart(2,'0'); const el = document.getElementById('currentTime'); if (el) el.textContent = `${m}:${s}`; }

function highlightActiveSegment(time) {
  let activeIdx = -1;
  for (let i = 0; i < segmentsData.length; i++) { if (time >= segmentsData[i].time && time < segmentsData[i].end) { activeIdx = i; break; } }
  if (activeIdx !== currentActiveSegment) {
    if (currentActiveSegment >= 0) { const prev = document.querySelector(`.segment-row[data-index="${currentActiveSegment}"]`); if (prev) { prev.classList.remove('active'); const ts = prev.querySelector('.timestamp'); if (ts) ts.classList.remove('active'); } }
    if (activeIdx >= 0) { const curr = document.querySelector(`.segment-row[data-index="${activeIdx}"]`); if (curr) { curr.classList.add('active'); const ts = curr.querySelector('.timestamp'); if (ts) ts.classList.add('active'); curr.scrollIntoView({ behavior: 'smooth', block: 'center' }); } }
    currentActiveSegment = activeIdx;
  }
}

function highlightSpeaker(speaker) {
  document.querySelectorAll('.segment-row').forEach(row => {
    const rs = row.querySelector('.speaker-tag').textContent.trim();
    row.style.opacity = (speaker === 'all' || rs === speaker) ? '1' : '0.3';
  });
}

    aiEl.className = 'flex gap-3 fade-in';

  const btn = document.getElementById('pipelineBtn');
  btn.disabled = true; btn.innerHTML = '⏳ Elaborazione...'; btn.classList.add('opacity-75','cursor-not-allowed');
  const log = document.getElementById('processingLog'); log.innerHTML = '';
  const steps = [
    { name:'upload', delay:500, status:'processing', msg:'Ricezione file audio: intercettazione_142.mp3 (45.2 MB)' },
    { name:'upload', delay:1000, status:'complete', msg:'Upload completato. Path: {user_id}/{case_id}/142/intercettazione_142.mp3' },

    { name:'chunking', delay:5000, status:'processing', msg:'Chunking: 500-800 caratteri, overlap 15% (75-120 chars)' },
    { name:'chunking', delay:6000, status:'complete', msg:'Chunking completato: 247 chunks creati.' },
    { name:'embedding', delay:6500, status:'processing', msg:'Generazione embedding via OpenAI text-embedding-3-large (1536d)' },
    { name:'embedding', delay:8000, status:'processing', msg:'Batch 124/247 → Vector(1536) generato' },
    { name:'embedding', delay:10000, status:'complete', msg:'247 embedding salvati in pgvector. IVFFlat index aggiornato.' },
    { name:'ready', delay:10500, status:'complete', msg:'✅ Pipeline completata. File pronto per query RAG.' }
  ];
  steps.forEach((step, i) => {
    setTimeout(() => {
      updateStepStatus(step.name, step.status);
      const ts = new Date().toISOString().substr(11,8);
      const color = step.status === 'complete' ? 'text-emerald-400' : step.status === 'processing' ? 'text-gold-500' : 'text-navy-500';
      const icon = step.status === 'complete' ? '✓' : step.status === 'processing' ? '⟳' : '•';
      log.innerHTML += `<p><span class="text-navy-600">[${ts}]</span> <span class="${color}">${icon}</span> ${step.msg}</p>`;
      log.scrollTop = log.scrollHeight;
      if (i === steps.length - 1) { btn.disabled = false; btn.innerHTML = '▶ Simula'; btn.classList.remove('opacity-75','cursor-not-allowed'); }
    }, step.delay);
  });
}
function updateStepStatus(name, status) {
  const el = document.getElementById(`step-${name}`);
  const sEl = document.getElementById(`status-${name}`);
  if (!el || !sEl) return;
  const border = el.querySelector('div'), icon = el.querySelector('svg');
  if (status === 'processing') { border.classList.remove('border-navy-700'); border.classList.add('border-gold-500'); icon.classList.remove('text-navy-500'); icon.classList.add('text-gold-500','animate-spin-slow'); sEl.className = 'badge status-processing'; sEl.textContent = 'In corso...'; }
  else if (status === 'complete') { border.classList.remove('border-gold-500'); border.classList.add('border-emerald-500'); icon.classList.remove('text-navy-500','text-gold-500','animate-spin-slow'); icon.classList.add('text-emerald-500'); sEl.className = 'badge status-complete'; sEl.textContent = 'Completato ✓'; }
}
